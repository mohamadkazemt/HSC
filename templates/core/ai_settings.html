راهنمای تنظیم Environment Variables

برای فعال‌سازی Multi-Provider Fallback:

# Google Gemini API Keys (comma-separated for key rotation)
GOOGLE_API_KEYS=key1,key2,key3
# Groq API Key
GROQ_API_KEY=your_groq_api_key
# OpenRouter API Key
OPENROUTER_API_KEY=your_openrouter_api_key
# Optional: Custom Models
GOOGLE_DEFAULT_MODEL=gemini-2.0-flash-lite
GROQ_MODEL=llama3-70b-8192
OPENROUTER_MODEL=google/gemini-2.0-flash-lite:free
